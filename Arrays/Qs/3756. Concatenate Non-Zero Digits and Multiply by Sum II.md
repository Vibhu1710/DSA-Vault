You are given a string `s` of length `m` consisting of digits. You are also given a 2D integer array `queries`, where `queries[i] = [l<sub>i</sub>, r<sub>i</sub>]`.

For each `queries[i]`, extract the **<button>substring</button>** `s[l<sub>i</sub>..r<sub>i</sub>]`. Then, perform the following:

* Form a new integer `x` by concatenating all the **non-zero digits** from the substring in their original order. If there are no non-zero digits, `x = 0`.
* Let `sum` be the **sum of digits** in `x`. The answer is `x * sum`.

Return an array of integers `answer` where `answer[i]` is the answer to the `i<sup>th</sup>` query.

Since the answers may be very large, return them **modulo** `10<sup>9</sup> + 7`.

**Example 1:**

**Input:** s = "10203004", queries = \[\[0,7\],\[1,3\],\[4,6\]\]

**Output:** \[12340, 4, 9\]

**Explanation:**

* `s[0..7] = "10203004"`

  * `x = 1234`
  * `sum = 1 + 2 + 3 + 4 = 10`
  * Therefore, answer is `1234 * 10 = 12340`.
* `s[1..3] = "020"`

  * `x = 2`
  * `sum = 2`
  * Therefore, the answer is `2 * 2 = 4`.
* `s[4..6] = "300"`

  * `x = 3`
  * `sum = 3`
  * Therefore, the answer is `3 * 3 = 9`.

**Example 2:**

**Input:** s = "1000", queries = \[\[0,3\],\[1,1\]\]

**Output:** \[1, 0\]

**Explanation:**

* `s[0..3] = "1000"`

  * `x = 1`
  * `sum = 1`
  * Therefore, the answer is `1 * 1 = 1`.
* `s[1..1] = "0"`

  * `x = 0`
  * `sum = 0`
  * Therefore, the answer is `0 * 0 = 0`.

**Example 3:**

**Input:** s = "9876543210", queries = \[\[0,9\]\]

**Output:** \[444444137\]

**Explanation:**

* `s[0..9] = "9876543210"`

  * `x = 987654321`
  * `sum = 9 + 8 + 7 + 6 + 5 + 4 + 3 + 2 + 1 = 45`
  * Therefore, the answer is `987654321 * 45 = 44444444445`.
  * We return `44444444445 modulo (10<sup>9</sup> + 7) = 444444137`.

**Constraints:**

* `1 <= m == s.length <= 10<sup>5</sup>`
* `s` consists of digits only.
* `1 <= queries.length <= 10<sup>5</sup>`
* `queries[i] = [l<sub>i</sub>, r<sub>i</sub>]`
* `0 <= l<sub>i</sub> <= r<sub>i</sub> < m`

```cpp
#define ll long long

#define M (ll)(1e9 + 7)
  

ll mod(ll a){return ((a%M + M ) % M);}

ll add(ll a, ll b){ return mod(mod(a) + mod(b));}

ll mul(ll a, ll b){ return mod(mod(a) * mod(b));}

ll sub(ll a, ll b){ return mod(mod(a) - mod(b) + M);}


class Solution {
public:
    vector<int> sumAndMultiply(string s, vector<vector<int>>& queries) {
        int m = s.size();

        vector<ll> tenPow(m + 1);
        tenPow[0] = 1;
        for(int i=0; i<m; ++i) tenPow[i+1] = mul(tenPow[i], 10);

        vector<ll> prefixSum(m+1);
        for(int i=0; i<m; ++i)
            prefixSum[i+1] = add(prefixSum[i], (s[i] - '0'));

        vector<pair<int, ll>> prefixNum(m+1);
        // the 0th element is a pair of (0, 0)
        for(int i=0; i<m; ++i){
            int dig = s[i] - '0';
            if(dig == 0) prefixNum[i+1] = prefixNum[i];
            else {
                prefixNum[i+1].first = prefixNum[i].first + 1;
                prefixNum[i+1].second = add(mul(prefixNum[i].second, 10), dig);
            }
        }

        vector<int> result;
        for(auto query: queries){
            int l = query[0];
            int r = query[1];

            // ans calculation..
            // x value extraction
            int nonZeroDigitCnt = prefixNum[r+1].first - prefixNum[l].first;
            ll x = sub(prefixNum[r+1].second, mul(prefixNum[l].second, tenPow[nonZeroDigitCnt]));
            // sum value extraction
            ll sum = sub(prefixSum[r+1], prefixSum[l]);
            result.push_back((int) mul(x, sum));
        }

        return result;
    }
};
```
