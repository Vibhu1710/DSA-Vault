You are given a **positive** integer `n`.

For every integer `x` from 1 to `n`, we write down the integer obtained by removing all zeros from the decimal representation of `x`.

Return an integer denoting the number of **distinct** integers written down.

**Example 1:**

**Input:** n = 10

**Output:** 9

**Explanation:**

The integers we wrote down are 1, 2, 3, 4, 5, 6, 7, 8, 9, 1. There are 9 distinct integers (1, 2, 3, 4, 5, 6, 7, 8, 9).

**Example 2:**

**Input:** n = 3

**Output:** 3

**Explanation:**

The integers we wrote down are 1, 2, 3. There are 3 distinct integers (1, 2, 3).

**Constraints:**

* `1 <= n <= 10<sup>15</sup>`

#include <bits/stdc++.h>
using namespace std;

/*
 Count how many integers in [1, n] do NOT contain the digit 0.
 n fits in 64-bit (n <= 1e15 per problem statement), so use long long.
*/

long long countNoZero(long long n) {
    string s = to_string(n);
    int L = s.size();

    // precompute powers of 9: pow9[k] = 9^k
    vector<long long> pow9(L+1, 1);
    for (int i = 1; i <= L; ++i) pow9[i] = pow9[i-1] * 9LL;

    long long ans = 0;

    // 1) Numbers with length < L
    for (int k = 1; k < L; ++k) {
        ans += pow9[k]; // 9^k numbers of length k (digits 1..9 each)
    }

    // 2) Numbers with length == L and <= n
    // process each digit, counting choices that are strictly smaller at the first differing position
    for (int i = 0; i < L; ++i) {
        int d = s[i] - '0';
        if (d == 0) {
            // no way to place a non-zero digit less than 0 -> stop
            break;
        }
        // place any non-zero digit less than d at this position
        // there are (d-1) choices (digits 1..d-1). For each, remaining positions have 9^(L-i-1) options.
        ans += (long long)(d - 1) * pow9[L - i - 1];

        // if we place exactly digit d (which is non-zero), we continue the tight prefix.
        // if i==L-1 and we never encountered a zero, we should include n itself:
        if (i == L - 1) {
            // we've matched all digits and none were zero => n itself is a valid zero-free number
            ans += 1;
        }
    }
    return ans;
}

    cin.tie(nullptr);

    long long n;
    if (!(cin >> n)) return 0;
    cout << countNoZero(n) << '\n';
    return 0;
}
